===============================================================
 MOPH NOTIFY API SPECIFICATION & PROMPT GUIDE
===============================================================

---------------------------------------------------------------
1. API SPECIFICATION (MOPH NOTIFY / หมอพร้อม)
---------------------------------------------------------------
Endpoint URL : https://morpromt2f.moph.go.th/api/notify/send
HTTP Method  : POST

Headers:
  Content-Type: application/json
  client-key: <CLIENT_KEY>
  secret-key: <SECRET_KEY>

Request Body (JSON format):
{
  "messages": [
    {
      "type": "text",
      "text": "ข้อความที่ต้องการแจ้งเตือน เช่น แจ้งเตือนจากระบบ..."
    }
  ]
}

HTTP Response:
  - 200 หรือ 201 : ส่งข้อความสำเร็จ
  - อื่นๆ (400, 401, 500) : ไม่สำเร็จ เช่น Key ไม่ถูกต้อง หรือรูปแบบ JSON ผิด

[PHP cURL]
---------------------------------------------------------------
function sendMophNotify($message, $clientKey, $secretKey) {
    $url = "https://morpromt2f.moph.go.th/api/notify/send";
    $payload = json_encode([
        "messages" => [
            ["type" => "text", "text" => $message]
        ]
    ], JSON_UNESCAPED_UNICODE);

    $ch = curl_init($url);
    curl_setopt($ch, CURLOPT_POST, true);
    curl_setopt($ch, CURLOPT_POSTFIELDS, $payload);
    curl_setopt($ch, CURLOPT_RETURNTRANSFER, true);
    curl_setopt($ch, CURLOPT_SSL_VERIFYPEER, false);
    curl_setopt($ch, CURLOPT_HTTPHEADER, [
        "Content-Type: application/json",
        "client-key: " . $clientKey,
        "secret-key: " . $secretKey
    ]);

    $response = curl_exec($ch);
    $httpCode = curl_getinfo($ch, CURLINFO_HTTP_CODE);
    $error    = curl_error($ch);
    curl_close($ch);

    if ($error) {
        return ['status' => false, 'error' => $error];
    }
    return ['status' => in_array($httpCode, [200, 201]), 'code' => $httpCode, 'response' => $response];
}

[Python]

import requests

def send_moph_notify(message, client_key, secret_key):
    url = "https://morpromt2f.moph.go.th/api/notify/send"
    headers = {
        "Content-Type": "application/json",
        "client-key": client_key,
        "secret-key": secret_key
    }
    data = {
        "messages": [
            {"type": "text", "text": message}
        ]
    }
    res = requests.post(url, json=data, headers=headers, verify=False)
    return res.status_code in [200, 201], res.text

async function sendMophNotify(message, clientKey, secretKey) {
  const url = "https://morpromt2f.moph.go.th/api/notify/send";
  const res = await fetch(url, {
    method: "POST",
    headers: {
      "Content-Type": "application/json",
      "client-key": clientKey,
      "secret-key": secretKey
    },
    body: JSON.stringify({
      messages: [{ type: "text", text: message }]
    })
  });
  return res.ok;
}
